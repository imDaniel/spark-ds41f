"""Replay this observation from archived CUDA ELF bytes, without a GPU.

Usage: python verify-machine-code.py.txt <hardware-evidence-directory>
This is an archival analysis driver, not a maintained release gate.
"""
import hashlib
import json
import pathlib
import struct
import sys
import tarfile

root = pathlib.Path(sys.argv[1])
manifest = json.loads((root / 'stock-mhc-artifacts.json').read_text())
expected = json.loads((root / 'stock-mhc-machine-code-identity.json').read_text())
archive = root / manifest['archive']
assert hashlib.sha256(archive.read_bytes()).hexdigest() == manifest['archive_sha256']


def execution(raw):
    assert raw[:7] == b'\x7fELF\x02\x01\x01'
    h = struct.unpack_from('<HHIQQQIHHHHHH', raw, 16)
    assert h[1] == 190 and h[10] == 64
    sections = [struct.unpack_from('<IIQQQQIIQQ', raw, h[5] + i*h[10]) for i in range(h[11])]
    name_header = sections[h[12]]
    names = raw[name_header[4]:name_header[4]+name_header[5]]
    code, constants, shared = [], [], []
    for s in sections:
        name = names[s[0]:].split(b'\0', 1)[0].decode()
        data = raw[s[4]:s[4]+s[5]]
        if name.startswith('.text.'):
            code.append(dict(name=name, bytes=s[5], sha256=hashlib.sha256(data).hexdigest(), flags=s[2], info=s[7], alignment=s[8]))
        elif name.startswith('.nv.constant'):
            constants.append(dict(name=name, bytes=s[5], sha256=hashlib.sha256(data).hexdigest(), flags=s[2], alignment=s[8]))
        elif name.startswith('.nv.shared'):
            shared.append(dict(name=name, bytes=s[5], flags=s[2], alignment=s[8]))
    assert code
    return dict(cuda_elf_flags=h[6], code=code, constants=constants, shared=shared)


observed = {}
with tarfile.open(archive) as tar:
    assert len(tar.getnames()) == len(set(tar.getnames()))
    for item in manifest['entries']:
        raw = tar.extractfile(item['member']).read()
        assert hashlib.sha256(raw).hexdigest() == item['sha256']
    for arm, rows in expected['rows'].items():
        observed[arm] = []
        for row in rows:
            raw = tar.extractfile(arm + '-' + row['sha256'] + '.cubin').read()
            assert hashlib.sha256(raw).hexdigest() == row['sha256']
            data = execution(raw)
            digest = hashlib.sha256(json.dumps(data, sort_keys=True).encode()).hexdigest()
            assert data == row['execution'] and digest == row['execution_sha256']
            observed[arm].append(digest)
a, b = set(observed['official']), set(observed['candidate'])
assert len(a) == expected['official_kernels'] == 6
assert len(b) == expected['candidate_kernels'] == 7
assert not a-b and expected['missing_stock_execution'] == 0
print(json.dumps(dict(official_kernels=len(a), candidate_kernels=len(b), all_stock_execution_equal=True)))
