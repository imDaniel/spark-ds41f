"""Run identically in independent parent/candidate images; compare JSON hashes."""
import argparse
import gc
import hashlib
import json
import statistics
import time
from pathlib import Path
from types import SimpleNamespace

import torch

from vllm.models.deepseek_v4_1.tilelang import mhc
from vllm.models.deepseek_v4_1.tilelang.collectives import RankOrderParts
from vllm.v1.worker.workspace import (
    collect_cuda_graph_capture_resources,
    init_workspace_manager,
    reset_workspace_manager,
)

parser = argparse.ArgumentParser()
parser.add_argument('--arm', required=True)
parser.add_argument('--output', type=Path, required=True)
args = parser.parse_args()
args.output.mkdir(parents=True, exist_ok=True)
assert torch.cuda.get_device_capability() == (12, 1)
init_workspace_manager(torch.device('cuda'))


def digest(tensor):
    return hashlib.sha256(tensor.contiguous().view(torch.uint8).cpu().numpy().tobytes()).hexdigest()


record = {
    'arm': args.arm,
    'module': mhc.__file__,
    'mhc_source_sha256': hashlib.sha256(Path(mhc.__file__).read_bytes()).hexdigest(),
    'torch': torch.__version__,
    'device': torch.cuda.get_device_name(),
    'scope': 'isolated stock mHC correctness and diagnostic timings; no model TPS claim',
    'cases': [],
}
start = time.monotonic()
try:
    for rows in (1, 96, 8192):
        for mode, world in (('pre', 1), ('broadcast', 1), ('post_pre', 1), ('post_pre', 4)):
            key = f'{mode}-world{world}-rows{rows}'
            seed = 1729 + rows + world * 1000000 + {'pre': 1, 'broadcast': 2, 'post_pre': 3}[mode] * 10000
            generator = torch.Generator(device='cuda').manual_seed(seed)
            h = 5120
            layer = mhc.TileKernelsMHC(SimpleNamespace(
                hc_mult=4, hidden_size=h, rms_norm_eps=1e-6, hc_eps=1e-6, hc_sinkhorn_iters=20,
            ))
            residual = torch.randn(rows, 4, h, generator=generator, device='cuda', dtype=torch.bfloat16)
            writer = torch.randn(world, rows, h, generator=generator, device='cuda', dtype=torch.bfloat16)
            post = torch.rand(rows, 4, generator=generator, device='cuda')
            comb = torch.rand(rows, 4, 4, generator=generator, device='cuda')
            pre = torch.rand(rows, 4, generator=generator, device='cuda')
            fn = torch.randn(24, 4 * h, generator=generator, device='cuda') / (4 * h) ** 0.5
            if mode == 'broadcast':
                fn = fn.view(24, 4, h).sum(dim=1).contiguous()
            scale = torch.ones(3, device='cuda')
            base = torch.zeros(24, device='cuda')
            norm = torch.ones(h, device='cuda', dtype=torch.bfloat16)
            inputs = dict(residual=residual, writer=writer, post=post, comb=comb, pre=pre, fn=fn)
            input_hashes = {name: digest(tensor) for name, tensor in inputs.items()}
            x = RankOrderParts(writer) if world == 4 else writer[0]

            def invoke():
                common = (fn, scale, base, norm, pre)
                if mode == 'pre':
                    return layer.pre(residual, *common)
                if mode == 'broadcast':
                    return layer.pre(writer[0], *common)
                return layer.post_pre(x, residual, post, comb, *common)

            stream = torch.cuda.Stream()
            stream.wait_stream(torch.cuda.current_stream())
            with torch.cuda.stream(stream):
                for _ in range(3):
                    warm = invoke()
            torch.cuda.current_stream().wait_stream(stream)
            torch.cuda.synchronize()
            graph = torch.cuda.CUDAGraph()
            with collect_cuda_graph_capture_resources() as resources, torch.cuda.graph(graph):
                observed = invoke()
            pointers = [tensor.data_ptr() for tensor in observed]
            graph.replay()
            expected = [digest(tensor) for tensor in observed]
            for _ in range(8):
                graph.replay()
            assert expected == [digest(tensor) for tensor in observed]
            assert pointers == [tensor.data_ptr() for tensor in observed]
            assert resources
            timings = []
            replays = 100 if rows == 8192 else 1000
            for _ in range(7):
                begin, end = torch.cuda.Event(enable_timing=True), torch.cuda.Event(enable_timing=True)
                begin.record()
                for _ in range(replays):
                    graph.replay()
                end.record()
                end.synchronize()
                timings.append(begin.elapsed_time(end) * 1000 / replays)
            with torch.profiler.profile(activities=[torch.profiler.ProfilerActivity.CPU, torch.profiler.ProfilerActivity.CUDA]) as profile:
                graph.replay()
                torch.cuda.synchronize()
            profile.export_chrome_trace(str(args.output / f'{args.arm}-{key}-trace.json'))
            row = {
                'case': key, 'rows': rows, 'world': world, 'mode': mode, 'seed': seed,
                'input_sha256': input_hashes, 'output_sha256': expected,
                'output_shapes': [list(t.shape) for t in observed],
                'graph_replays_bit_stable': True,
                'cuda_graph_us': timings, 'median_us': statistics.median(timings),
            }
            record['cases'].append(row)
            (args.output / f'{args.arm}.json').write_text(json.dumps(record, indent=2) + '\n')
            print(json.dumps({'arm': args.arm, 'case': key, 'median_us': row['median_us'], 'graph_bit_stable': True}), flush=True)
            del graph, resources, observed, warm, layer, inputs, residual, writer, x, post, comb, pre, fn
            gc.collect()
            torch.cuda.empty_cache()
finally:
    record['seconds'] = time.monotonic() - start
    (args.output / f'{args.arm}.json').write_text(json.dumps(record, indent=2) + '\n')
    reset_workspace_manager()
